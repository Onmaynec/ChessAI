cmake_minimum_required(VERSION 3.16)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" CHESSAI_VERSION LIMIT_COUNT 1)
project(ChessAI VERSION ${CHESSAI_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(chess_core src/chess.cpp src/search.cpp)
target_include_directories(chess_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(MSVC)
  target_compile_options(chess_core PRIVATE /W4 /O2)
else()
  target_compile_options(chess_core PRIVATE -Wall -Wextra -Wpedantic -O3)
endif()
add_executable(chessai src/main.cpp)
target_link_libraries(chessai PRIVATE chess_core Threads::Threads)
add_executable(chessai_tests tests/tests.cpp)
target_link_libraries(chessai_tests PRIVATE chess_core Threads::Threads)
enable_testing()
add_test(NAME chess_core_tests COMMAND chessai_tests)
